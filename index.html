  3. Pseudo-selector

    :has()

    :has() acts as a parent selector - style the parent depending on the child(ren) 

                          
                        .dynamic-form {
                        container: dynamic-form;
                        container-type: inline-size;
                        --unstyled: 0;
                        --border-color: var(--theme-color);
                        }

                        .dynamic-form:has(#ready:not(:checked)) {
                        --unstyled: 1;
                        --border-color: transparent;
                        }
